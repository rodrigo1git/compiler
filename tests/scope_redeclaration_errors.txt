// Repeating a variable in the same scope is an error.
integer a;

// A function cannot reuse a variable name from the same scope.
integer function variable(integer p)
integer isolated;
begin
    isolated := p;
    ret(isolated);
end;

// Parameters and local variables share a scope.
integer function repeated(integer p, integer p)
integer p;
begin
    ret(p);
end;

// Duplicate functions must not merge their parameters or local declarations.
integer function repeated(integer q)
integer isolated;
begin
    isolated := q;
    ret(isolated);
end;
