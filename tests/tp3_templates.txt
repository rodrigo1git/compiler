template_program
// Used templates keep their original symbols alongside their concrete bodies.
integer function display(comptime integer p)
begin
    pout(p);
    ret(p + 1$i);
end;
// An uninvoked template still shows its loop, selection and local jump targets.
integer function unused(comptime integer p)
integer i;
begin
    from i := 0$i to 2$i by 1$i repeat begin
        if (p > i) pout(i); end_if;
    end;
    ret(p);
end;
class box code
begin
    integer value;
    integer read(comptime integer p)
    begin
        ret(value + p);
    end;
    // An uninvoked method also has a symbolic body, without inventing p's value.
    integer unused(comptime integer p)
    begin
        pout(p);
    end;
end;
box object;
begin
    // Concrete output: 3, 4, 7, 8, 6. Unused bodies produce no output.
    pout(display(3$i)[1$i]);
    pout(display(7$i)[1$i]);
    object[0$i] := 5$i;
    pout(object.read(1$i)[1$i]);
end;
