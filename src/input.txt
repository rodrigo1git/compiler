integer a, b, contador;

integer function suma(integer x, integer y)
    integer aux;
begin
    aux := x + y;
    ret aux;
end;

class punto
begin
    px := 0$i;
    py := -0$i;

    integer function norma(integer p)
        integer a;
    begin
        ret p;
    end;
end;

begin
    a := 10$i;
    b := 32$i;
    promedio := -1.5s+2;          // notacion cientifica: 1.5 * 10^2
    contador := suma(a, b);

    if (a >= b) begin
        pout("a es mayor o igual");
    end
    else
        pout("b es mayor");
    end_if

    // el ; final cierra el repeat (el cuerpo trae el suyo propio)
    from contador := 1$i to 10$i by 1$i repeat

    ;

    toi(promedio);
